2. 修改 JavaScript：建立權限清單並分類顯示
在 dashboard.js 中，我們可以建立一個對照表（Mapping），定義哪些 Email 擁有哪些選單的開啟權限。

JavaScript
// dashboard.js 中的 onAuthStateChanged 部分
auth.onAuthStateChanged((user) => {
    if (user) {
        // --- 1. 定義權限清單 ---
        const permissions = {
            "upload": ["admin_a@ll.com", "boss@ll.com"], // 可看上傳檔案的人
            "event": ["admin_b@ll.com", "boss@ll.com"]   // 可看新增活動的人
        };

        // --- 2. 取得 DOM 元素 ---
        const uploadBtn = document.getElementById('nav-upload-files');
        const eventBtn = document.getElementById('nav-add-event');

        // --- 3. 判斷並顯示 (使用 includes 檢查當前登入者 Email) ---
        if (uploadBtn && permissions.upload.includes(user.email)) {
            uploadBtn.style.display = "block"; 
        }
        
        if (eventBtn && permissions.event.includes(user.email)) {
            eventBtn.style.display = "block"; 
        }

        // ... 其餘邏輯維持不變 ...
    } else {
        window.location.href = "login.html";
    }
});
