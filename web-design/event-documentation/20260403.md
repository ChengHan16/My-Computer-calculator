        function startPublicEventListen() {
            const display = document.getElementById('public-event-display');
            db.collection("events").onSnapshot(snapshot => {
                display.innerHTML = "";
                let eventsArray = [];
                
                snapshot.forEach(doc => {
                    const ev = doc.data();
                    const id = doc.id;
                    const timeField = ev.fields ? ev.fields.find(f => f.type === 'time') : null;
                    let sortDate = timeField ? new Date(timeField.content) : (ev.createdAt ? ev.createdAt.toDate() : new Date());
                    eventsArray.push({ id, ...ev, sortDate });
                });

                eventsArray.sort((a, b) => b.sortDate - a.sortDate);

                let currentYear = null;
                eventsArray.forEach(ev => {
                    const year = ev.sortDate.getFullYear();
                    if (year !== currentYear) {
                        currentYear = year;
                        display.insertAdjacentHTML('beforeend', `<div class="year-group-title">${year} 年度活動紀實</div>`);
                    }

                    const countdownText = ev.deadline ? getCountdownText(ev.deadline) : "";
                    const countdownHtml = countdownText ? `<span class="countdown-alert">${countdownText}</span>` : "";

                        <div class="doc-list-item" onclick="openDocPreview('${ev.id}')">

            function loadDynamicScript(src, callback) {
                // 避免重複載入相同腳本
                if (document.querySelector(`script[src="${src}"]`)) {
                    if (callback) callback();
                    return;
                }
                const script = document.createElement('script');
                script.src = src;
                script.onload = callback;
                document.body.appendChild(script);

                // 從 act 集合抓取該帳號資料
                const userDoc = await db.collection("act").doc(user.uid).get();
                const userData = userDoc.data() || {};

                await db.collection("events").doc(currentEventId).collection("votes").doc(user.uid).set({
                    name: userData.displayName || user.email.split('@')[0],
                    photo: userData.photoURL || "",
                    choice: choice,
                    timestamp: firebase.firestore.FieldValue.serverTimestamp()
                });

                alert(choice === 'join' ? "已記錄您的參與狀況：參與" : "已記錄您的參與狀況：下次再聚");
                loadVotes(currentEventId);
            } catch (error) {
                console.error("投票失敗:", error);
            }
        }

        function loadVotes(eventId) {
            const list = document.getElementById('voter-list');
            db.collection("events").doc(eventId).collection("votes")
              .where("choice", "==", "join")
              .onSnapshot(snapshot => {
                list.innerHTML = "";
                if (snapshot.empty) {
                    list.innerHTML = "<p style='color:#ccc; font-size:0.9rem;'>尚無成員參與...</p>";
                    return;
                }
                snapshot.forEach(doc => {
                    const v = doc.data();
                    list.innerHTML += `
                        <div class="voter-box">
                            <img src="${v.photo || 'https://via.placeholder.com/60'}" onerror="this.src='https://via.placeholder.com/60'">
                            <span>${v.name}</span>
                        </div>`;
                });
            });
        }

        /**
         * 核心功能：計算並格式化倒數時間
         */
        function getCountdownText(deadlineStr) {
            if (!deadlineStr) return "";
            
            const now = new Date().getTime();
            const deadline = new Date(deadlineStr).getTime();
            const diff = deadline - now;

            if (diff <= 0) return "投票已截止";

            const days = Math.floor(diff / (1000 * 60 * 60 * 24));
            const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

            // 如果超過一天，顯示 Day X | XX 小時 XX 分
            if (days >= 1) {
                return `投票倒數 Day ${days} | ${hours.toString().padStart(2, '0')} 小時 ${minutes.toString().padStart(2, '0')} 分`;
            } 
            // 如果小於一天，顯示 XX 小時 XX 分
            else {
                return `投票倒數 ${hours.toString().padStart(2, '0')} 小時 ${minutes.toString().padStart(2, '0')} 分`;
            }
        }
